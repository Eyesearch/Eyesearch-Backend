"""The six pipeline agents.

Each agent is a numbered module — `01_frontier_agent` … `06_revalidation_agent`
— matching the brief's file names. Python cannot import a module whose name
starts with a digit using `import` syntax, so the modules are reached through
`load_agent()` here, which uses the import machinery's string form.

Nothing in this package imports the agents at import time: that keeps
`from app.agents import load_agent` free of circular imports, since every agent
module can then import this package safely.
"""

from __future__ import annotations

from importlib import import_module
from types import ModuleType

__all__ = ["AGENT_MODULES", "load_agent"]

#: Every agent module, in pipeline order.
AGENT_MODULES: tuple[str, ...] = (
    "01_frontier_agent",
    "02_fetcher_agent",
    "03_parser_agent",
    "04_quality_agent",
    "05_indexer_agent",
    "06_revalidation_agent",
    "07_classifier_agent",
    "08_language_agent",
)


def load_agent(name: str) -> ModuleType:
    """Import one numbered agent module by its file name."""
    return import_module(f"{__name__}.{name}")

"""Agent 01 — the frontier.

Owns `crawl_queue`: what is waiting to be crawled, in what order, and which
domains are worth spending a request on first.

Queue order is the primary key (`rowid`), exactly as the pipeline reads it, so
priority has to be applied when the URL is enqueued rather than when it is
dequeued. Official Indian public domains therefore go in ahead of everything
else, and the dequeue query stays a plain `ORDER BY rowid`.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from urllib.parse import parse_qsl, urlparse, urlunparse

from app.engine.config import Settings, get_settings
from app.engine.store import IndexStore, QueuedUrl

logger = logging.getLogger(__name__)

)

        ".css", ".js", ".mjs", ".json", ".xml", ".rss", ".atom", ".txt",

#: Query keys that identify a campaign, never a different page.

@dataclass(frozen=True)
class FrontierItem:
    """A URL the frontier accepted, with the reason it ranks where it does."""

    url: str
    domain: str
    priority: int


def domain_of(url: str) -> str:
    """The registrable host of a URL, lower-cased and without `www.`."""
    host = (urlparse(url).hostname or "").lower()
    return host[4:] if host.startswith("www.") else host


def normalise_url(url: str) -> str | None:
    """One canonical form per page, or `None` when it is not a page.

    Lower-case host, no `www.`, no fragment, no trailing slash, sorted query
    keys, tracking parameters removed. Two links that address the same page
    therefore collapse to one queue row, which is what stops the frontier
    filling with `?utm_source=` variants of the same URL.
    """
    if not url:
        return None

    parts = urlparse(url.strip())
    if parts.scheme not in ("http", "https"):
        return None

    host = (parts.hostname or "").lower()
    if not host:
        return None
    if host.startswith("www."):
        host = host[4:]

    path = parts.path or "/"
    if len(path) > 1:
        path = path.rstrip("/")

    query = "&".join(
        f"{key}={value}"
        for key, value in sorted(parse_qsl(parts.query, keep_blank_values=True))
        if key.lower() not in TRACKING_PARAMS
    )

    port = f":{parts.port}" if parts.port and parts.port not in (80, 443) else ""
    return urlunparse((parts.scheme, f"{host}{port}", path, "", query, ""))


def is_crawlable(url: str) -> bool:
    """Crawl-scope rules: depth, extension, and form/session paths."""
    parts = urlparse(url)
    path = (parts.path or "/").lower()

    if path.endswith(tuple(JUNK_EXTENSIONS)):
        return False

    depth = len([segment for segment in path.split("/") if segment])
    if depth > MAX_PATH_DEPTH:
        return False

    if any(marker in path for marker in BLOCKED_PATH_MARKERS):
        return False

    # A single unbroken key longer than a URL segment is a session id.
    if any(len(segment) > 80 for segment in path.split("/")):
        return False

    return True


def in_crawl_scope(url: str, source_host: str) -> bool:
    """Same host, or an authoritative Indian public domain."""
    host = domain_of(url)
    if not host:
        return False
    if host == source_host:
        return True
    return is_authoritative(host)


def is_authoritative(host: str) -> bool:
    """An Indian public suffix, or one of the whitelisted portals."""
    if any(host.endswith(suffix) for suffix in PRIORITY_SUFFIXES):
        return True
    return any(host == domain or host.endswith(f".{domain}") for domain in WHITELISTED_DOMAINS)


def priority_of(url: str) -> int:
    """Lower sorts first: official public domains, then everything else."""
    return 0 if is_authoritative(domain_of(url)) else 1


class FrontierAgent:
    """Reads and writes the crawl frontier."""

    def __init__(self, store: IndexStore, settings: Settings | None = None) -> None:
        self._store = store
        self._settings = settings or get_settings()

    async def enqueue(self, urls: list[str] | tuple[str, ...] | set[str]) -> int:
        """Queue URLs, priority first, skipping anything already known.

        Insert order is what makes priority real, so the list is sorted before
        it reaches the store rather than relying on the dequeue query.
        """
        candidates: list[FrontierItem] = []
        seen: set[str] = set()
        for url in urls:
            cleaned = (url or "").strip()
            if not cleaned.startswith(("http://", "https://")):
                continue
            if cleaned in seen:
                continue
            domain = domain_of(cleaned)
            if not domain:
                continue
            seen.add(cleaned)
            candidates.append(FrontierItem(cleaned, domain, priority_of(cleaned)))

        if not candidates:
            return 0

        pending = await self.depth()
        if pending >= MAX_PENDING_QUEUE:
            logger.debug(
                "frontier full (%d pending >= %d), dropping %d new URLs",
                pending,
                MAX_PENDING_QUEUE,
                len(candidates),
            )
            return 0

        # One call must not overshoot the cap: trim to the room that is left.
        capacity = MAX_PENDING_QUEUE - pending
        if len(candidates) > capacity:
            logger.debug(
                "frontier near capacity (%d left), keeping %d of %d URLs",
                capacity,
                capacity,
                len(candidates),
            )
            candidates = candidates[:capacity]

        candidates.sort(key=lambda item: (item.priority, len(item.url), item.url))
        known = await self._store.known_urls([item.url for item in candidates])
        fresh = [item for item in candidates if item.url not in known]
        if not fresh:
            return 0

        added = await self._store.enqueue_urls([(item.url, item.domain) for item in fresh])
        if added:
            logger.info("frontier: queued %d new URLs (depth now %d)", added, await self.depth())
        return added

    def filter_links(self, links: list[str] | tuple[str, ...], source_url: str) -> list[str]:
        """The subset of a page's links that belongs in the frontier.

        Pure and synchronous: no database, no network. The orchestrator calls
        this per page and enqueues the aggregate once per batch.
        """
        source_host = domain_of(source_url)
        accepted: list[str] = []
        seen: set[str] = set()

        for link in links:
            normalised = normalise_url(link)
            if not normalised or normalised in seen:
                continue
            if normalised == normalise_url(source_url):
                continue
            if not is_crawlable(normalised):
                continue
            if not in_crawl_scope(normalised, source_host):
                continue
            seen.add(normalised)
            accepted.append(normalised)

        return accepted

    async def discover(self, links: list[str] | tuple[str, ...], source_url: str) -> int:
        """Filter a page's links and queue the survivors."""
        candidates = self.filter_links(links, source_url)
        if not candidates:
            return 0
        return await self.enqueue(candidates)

    async def next_batch(self, limit: int | None = None) -> list[QueuedUrl]:
        """The next batch of pending URLs, in the order the plan specifies."""
        size = limit or self._settings.crawl_batch_size
        return await self._store.pending_batch(size)

    async def depth(self) -> int:
        return await self._store.queue_depth()

    async def refill(self, minimum: int = 1) -> int:
        """Keep the frontier alive when it runs dry.

        Two stages, in order:

        1. queue any seed that has not been seen yet;
        2. if every seed is already indexed, return the stalest crawled URLs to
           `pending` once their cooldown has elapsed.

        Stage 2 is what stops the stall: with only stage 1, an exhausted seed
        list means `enqueue()` rejects every URL as already known, the queue
        stays empty, and the worker idles for good.
        """
        if await self.depth() >= minimum:
            return 0

        added = await self.enqueue(SEED_URLS)
        if added:
            return added

        rearmed = await self._store.rearm_processed(
            limit=self._settings.rearm_batch_size,
            cooldown_seconds=self._settings.rearm_cooldown_seconds,
        )
        if rearmed:
            logger.info(
                "frontier re-armed: %d previously crawled URLs returned to pending "
                "after a %ds cooldown",
                rearmed,
                self._settings.rearm_cooldown_seconds,
            )
        return rearmed

"""Agent 03 — the parser.

Turns HTML into the three fields the index stores: title, description and body
text. The body is **hard-capped at 1,000 characters**, cut at the nearest word
boundary and closed with an ellipsis, so a page can never push a row past the
storage budget and a card never shows half a word.
"""

from __future__ import annotations

import logging
import re
from dataclasses import dataclass, field
from urllib.parse import urljoin

from bs4 import BeautifulSoup

logger = logging.getLogger(__name__)

#: Hard ceiling on stored body text, ellipsis included.
BODY_CHAR_CAP = 1000

#: Ceilings for the two metadata fields.
TITLE_CHAR_CAP = 200
DESCRIPTION_CHAR_CAP = 400

#: A meta description shorter than this says nothing useful.
MIN_DESCRIPTION_CHARS = 20

#: How much body text a rebuilt description borrows.
DESCRIPTION_SNIPPET_CHARS = 180

#: Openings that mean the description is the site's boilerplate, repeated
#: verbatim on every subpage — so it identifies nothing about this page.
GENERIC_DESCRIPTION_PREFIXES: tuple[str, ...] = (
    "official portal",
    "official website",
    "official web site",
    "welcome to",
    "home page",
    "homepage",
    "this is the official",
    "government of india",
)

ELLIPSIS = "…"

#: How many outbound links one page may contribute to the frontier.
MAX_LINKS_PER_PAGE = 40

#: Hrefs that are never a page to crawl.
SKIP_HREF_PREFIXES: tuple[str, ...] = (
    "#",
    "mailto:",
    "tel:",
    "javascript:",
    "data:",
    "ftp:",
    "file:",
)

# Elements that are never prose.
STRIP_TAGS: tuple[str, ...] = (
    "script",
    "style",
    "noscript",
    "iframe",
    "svg",
    "canvas",
    "form",
    "nav",
    "footer",
    "header",
    "aside",
    "template",
)

WHITESPACE = re.compile(r"\s+")


def collapse(text: str) -> str:
    """Collapse every run of whitespace to one space and trim."""
    return WHITESPACE.sub(" ", (text or "").replace("\u00a0", " ")).strip()


def snippet_from(text: str, limit: int = DESCRIPTION_SNIPPET_CHARS) -> str:
    """The first `limit` characters of `text`, cut at a word boundary.

    The ellipsis is only appended when something was actually removed, so a
    short page does not advertise a truncation that did not happen.
    """
    cleaned = collapse(text)
    if len(cleaned) <= limit:
        return cleaned

    window = cleaned[:limit]
    boundary = window.rsplit(" ", 1)[0].rstrip(" ,;:-—")
    return (boundary or window) + ELLIPSIS


def is_generic_description(description: str, title: str) -> bool:
    """True when the meta description cannot distinguish this page."""
    cleaned = collapse(description)
    if len(cleaned) < MIN_DESCRIPTION_CHARS:
        return True

    lowered = cleaned.lower()
    if lowered.startswith(GENERIC_DESCRIPTION_PREFIXES):
        return True

    # A description that only repeats the title adds nothing to it.
    if title and lowered == collapse(title).lower():
        return True

    return False


def build_description(title: str, meta_description: str, body_text: str) -> str:
    """A description that is specific to this page.

    Portals often ship one boilerplate meta description across hundreds of
    subpages, which makes every result card read identically. When the meta
    description is missing, too short, or generic, the title is combined with
    the opening of the page's own body text instead.
    """
    meta = collapse(meta_description)
    if meta and not is_generic_description(meta, title):
        return meta[:DESCRIPTION_CHAR_CAP]

    snippet = snippet_from(body_text)
    if title and snippet:
        return f"{title} — {snippet}"[:DESCRIPTION_CHAR_CAP]
    return (snippet or meta or collapse(title))[:DESCRIPTION_CHAR_CAP]


def truncate_body(text: str, cap: int = BODY_CHAR_CAP) -> str:
    """Cut `text` to at most `cap` characters at a word boundary.

    The ellipsis counts against the cap, so the returned string is never
    longer than `cap`. A single unbroken token longer than the cap is cut at
    the cap rather than being returned whole.
    """
    cleaned = collapse(text)
    if len(cleaned) <= cap:
        return cleaned

    window = cleaned[: max(1, cap - len(ELLIPSIS))]
    boundary = window.rsplit(" ", 1)[0].rstrip(" ,;:-—")
    if not boundary:
        boundary = window
    return boundary + ELLIPSIS


@dataclass
class ParsedPage:
    """What the parser found, and whether it is usable."""

    url: str
    ok: bool
    title: str = ""
    description: str = ""
    body_text: str = ""
    reason: str = ""
    links: list[str] = field(default_factory=list)

    @property
    def char_count(self) -> int:
        return len(self.body_text)


class ParserAgent:
    """HTML in, bounded fields out. Never raises on bad markup."""

    def __init__(self, body_char_cap: int = BODY_CHAR_CAP) -> None:
        self.body_char_cap = body_char_cap

    def parse(self, url: str, html: str) -> ParsedPage:
        if not html or not html.strip():
            return ParsedPage(url=url, ok=False, reason="empty_response")

        try:
            soup = BeautifulSoup(html, "html.parser")
        except Exception as error:  # bs4 is tolerant, but never fatal here
            logger.debug("parse failed for %s: %s", url, error)
            return ParsedPage(url=url, ok=False, reason="unparseable")

        for tag in soup.find_all(STRIP_TAGS):
            tag.decompose()

        title = self._title(soup)
        body_text = truncate_body(soup.get_text(" "), self.body_char_cap)
        description = build_description(title, self._description(soup), body_text)
        links = self.extract_links(soup, url)

        if not title and not body_text:
            return ParsedPage(url=url, ok=False, reason="no_content")

        return ParsedPage(
            url=url,
            ok=True,
            title=title[:TITLE_CHAR_CAP],
            description=description[:DESCRIPTION_CHAR_CAP],
            body_text=body_text,
            links=links,
        )

    @staticmethod
    def extract_links(
        soup: BeautifulSoup,
        base_url: str,
        limit: int = MAX_LINKS_PER_PAGE,
    ) -> list[str]:
        """Absolute outbound URLs from `<a href>`, in document order.

        Relative hrefs are resolved against the page's own URL, so the
        frontier receives only absolute URLs and never has to know where a
        link came from. Fragments, `mailto:`, `javascript:` and the rest are
        dropped here; the frontier applies the crawl-scope rules.
        """
        links: list[str] = []
        seen: set[str] = set()

        for anchor in soup.find_all("a", href=True):
            href = str(anchor.get("href") or "").strip()
            if not href or href.lower().startswith(SKIP_HREF_PREFIXES):
                continue

            absolute = urljoin(base_url, href)
            if not absolute.lower().startswith(("http://", "https://")):
                continue
            # The fragment identifies a position in a page, not another page.
            absolute = absolute.split("#", 1)[0]
            if absolute in seen:
                continue
            seen.add(absolute)
            links.append(absolute)
            if len(links) >= limit:
                break

        return links

    @staticmethod
    def _title(soup: BeautifulSoup) -> str:
        if soup.title and soup.title.string:
            title = collapse(soup.title.string)
            if title:
                return title
        for attribute, key in (
            ("property", "og:title"),
            ("name", "twitter:title"),
        ):
            meta = soup.find("meta", attrs={attribute: key})
            if meta and meta.get("content"):
                title = collapse(meta["content"])
                if title:
                    return title
        heading = soup.find("h1")
        if heading:
            return collapse(heading.get_text(" "))
        return ""

    @staticmethod
    def _description(soup: BeautifulSoup) -> str:
        for attribute, key in (
            ("name", "description"),
            ("property", "og:description"),
            ("name", "twitter:description"),
        ):
            meta = soup.find("meta", attrs={attribute: key})
            if meta and meta.get("content"):
                description = collapse(meta["content"])
                if description:
                    return description
        return ""

"""Agent 02 — the fetcher.

One pooled `httpx.AsyncClient` for the whole process, a desktop browser's
header set on every request, bounded concurrency, and a byte cap so a hostile
or accidental multi-gigabyte response cannot exhaust a 512 MB instance.

The User-Agent carries no bot token: several Indian government WAFs answer 403
or 406 to anything that announces itself as a crawler, and a plain desktop
Chrome profile is what gets the page.
"""

from __future__ import annotations

import asyncio
import logging
import time
from dataclasses import dataclass

import httpx

from app.engine.config import Settings, get_settings

logger = logging.getLogger(__name__)

BOUND_HEADERS: dict[str, str] = {
    "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8",
    "Accept-Language": "en-US,en;q=0.9",
    "Upgrade-Insecure-Requests": "1",
    "Sec-Fetch-Dest": "document",
    "Sec-Fetch-Mode": "navigate",
    "Sec-Fetch-Site": "none",
    "Sec-Fetch-User": "?1",
    "Cache-Control": "no-cache",
}

# Statuses that mean "this host will not talk to us" rather than "this page is
# missing". Retrying them is how a crawl stalls on one domain forever.
BLOCKING_STATUSES: frozenset[int] = frozenset({401, 403, 406, 429, 451})


@dataclass
class FetchResult:
    """The outcome of one fetch, success or failure."""

    url: str
    ok: bool
    status: int = 0
    html: str = ""
    error: str = ""
    elapsed_ms: int = 0
    blocked: bool = False

    @property
    def reason(self) -> str:
        if self.ok:
            return ""
        if self.blocked:
            return f"blocked_status:{self.status}"
        if self.status:
            return f"http_status:{self.status}"
        return self.error or "fetch_failed"


class FetcherAgent:
    """Bounded, pooled HTTP fetching."""

    def __init__(
        self,
        settings: Settings | None = None,
        client: httpx.AsyncClient | None = None,
        transport: httpx.AsyncBaseTransport | None = None,
    ) -> None:
        self._settings = settings or get_settings()
        self._owns_client = client is None
        self._client = client or httpx.AsyncClient(
            timeout=self._settings.crawl_timeout_seconds,
            follow_redirects=True,
            transport=transport,
            headers={
                **BOUND_HEADERS,
                "User-Agent": self._settings.crawl_user_agent,
            },
        )
        self._semaphore = asyncio.Semaphore(max(1, self._settings.crawl_concurrency))
        self.fetches = 0

    @property
    def client(self) -> httpx.AsyncClient:
        return self._client

    async def fetch(self, url: str, *, attempts: int = 2) -> FetchResult:
        """Fetch one URL, retrying only what is worth retrying."""
        started = time.monotonic()
        last_error = ""
        last_status = 0

        for attempt in range(1, max(1, attempts) + 1):
            async with self._semaphore:
                try:
                    self.fetches += 1
                    response = await self._client.get(url)
                except httpx.TimeoutException:
                    last_error = "timeout"
                except httpx.HTTPError as error:
                    last_error = f"http_error:{type(error).__name__}"
                else:
                    last_status = response.status_code
                    if response.status_code in BLOCKING_STATUSES:
                        logger.debug("fetch %s blocked with %d", url, response.status_code)
                        return FetchResult(
                            url=url,
                            ok=False,
                            status=response.status_code,
                            blocked=True,
                            error=f"blocked_status:{response.status_code}",
                            elapsed_ms=int((time.monotonic() - started) * 1000),
                        )
                    if response.status_code >= 400:
                        last_error = f"http_status:{response.status_code}"
                    else:
                        html = self._read_capped(response)
                        return FetchResult(
                            url=url,
                            ok=True,
                            status=response.status_code,
                            html=html,
                            elapsed_ms=int((time.monotonic() - started) * 1000),
                        )

            if attempt < attempts:
                await asyncio.sleep(0.5 * attempt)

        return FetchResult(
            url=url,
            ok=False,
            status=last_status,
            error=last_error or "fetch_failed",
            elapsed_ms=int((time.monotonic() - started) * 1000),
        )

    def _read_capped(self, response: httpx.Response) -> str:
        """Decode at most `crawl_max_body_bytes`, never the whole stream."""
        raw = response.content[: self._settings.crawl_max_body_bytes]
        encoding = response.encoding or "utf-8"
        try:
            return raw.decode(encoding, errors="replace")
        except (LookupError, UnicodeDecodeError):
            return raw.decode("utf-8", errors="replace")

    async def fetch_many(self, urls: list[str]) -> list[FetchResult]:
        """Fetch a batch concurrently, bounded by the semaphore."""
        if not urls:
            return []
        return list(await asyncio.gather(*(self.fetch(url) for url in urls)))

    async def aclose(self) -> None:
        if self._owns_client:
            await self._client.aclose()

"""Agent 04 — the quality gate.

A page earns a row only if it carries at least **80 characters** of real body
text and is not a known junk pattern (WAF interstitial, JavaScript shell, error
page). Body text is fingerprinted with **SHA-256**, so the same prose can never
enter the index twice under two URLs — and within a single batch the gate
rejects the repeat before it ever reaches the database.
"""

from __future__ import annotations

import hashlib
import logging
from dataclasses import dataclass

from app.agents import load_agent
from app.engine.store import PageDocument

logger = logging.getLogger(__name__)

# The parser owns the one definition of "collapse whitespace"; reaching it
# through the loader keeps a single implementation rather than a copy.
collapse = load_agent("03_parser_agent").collapse

#: Below this many characters of body text a page is not worth a row.
MIN_BODY_CHARS = 80

#: A title shorter than this is a filename or a placeholder, not a heading.
MIN_TITLE_CHARS = 3

#: Phrases that mean the page is an interstitial, not content.
JUNK_MARKERS: tuple[str, ...] = (
    "403 forbidden",
    "access denied",
    "just a moment",
    "checking your browser",
    "enable javascript",
    "javascript is required",
    "please turn on javascript",
    "attention required! | cloudflare",
    "request blocked",
    "captcha",
    "page not found",
    "404 not found",
)


def digest_of(text: str) -> str:
    """SHA-256 of the cleaned body text, hex encoded."""
    return hashlib.sha256(collapse(text).encode("utf-8")).hexdigest()


@dataclass
class QualityVerdict:
    """The gate's decision, with the rule that made it."""

    ok: bool
    reason: str = ""
    digest: str = ""
    score: float = 1.0


class QualityAgent:
    """The anti-junk gate and in-run deduplicator."""

    def __init__(self, min_body_chars: int = MIN_BODY_CHARS) -> None:
        self.min_body_chars = min_body_chars
        self._seen: set[str] = set()

    def reset(self) -> None:
        """Forget the digests seen so far — called at the start of a batch."""
        self._seen.clear()

    def check(self, document: PageDocument) -> QualityVerdict:
        """Decide whether one document may be stored."""
        body = collapse(document.body_text)
        title = collapse(document.title)

        if len(body) < self.min_body_chars:
            return QualityVerdict(
                ok=False,
                reason="low_text_density",
                score=0.0,
            )

        lowered = body[:600].lower()
        for marker in JUNK_MARKERS:
            if marker in lowered:
                return QualityVerdict(ok=False, reason="junk_page", score=0.0)

        if len(title) < MIN_TITLE_CHARS:
            return QualityVerdict(ok=False, reason="no_title", score=0.0)

        digest = digest_of(body)
        if digest in self._seen:
            return QualityVerdict(ok=False, reason="duplicate", digest=digest, score=0.0)

        self._seen.add(digest)
        return QualityVerdict(
            ok=True,
            digest=digest,
            score=self._score(body, document),
        )

    @staticmethod
    def _score(body: str, document: PageDocument) -> float:
        """A small quality signal: more prose and a description score higher."""
        length_score = min(1.0, len(body) / 1000.0)
        description_bonus = 0.1 if document.description.strip() else 0.0
        return round(min(1.0, 0.5 + 0.4 * length_score + description_bonus), 3)

"""Agent 05 — the indexer.

The only module that writes pages. It asks the storage guard once per batch,
refuses to write when the guard is paused, and reports a page count only after
the database has confirmed the commit.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from typing import Sequence

from app.engine.store import IndexStore, PageDocument, PersistOutcome
from app.indexer.guard import StorageGuard, StoragePaused, StorageState

logger = logging.getLogger(__name__)


@dataclass
class IndexOutcome:
    """What the indexer achieved, in the terms the orchestrator reports."""

    committed: bool
    pages_written: int = 0
    duplicates: int = 0
    queue_closed: int = 0
    total_pages: int = 0
    reason: str = ""
    paused: bool = False

    @property
    def counted(self) -> int:
        """Pages the caller may add to its running total.

        Only a confirmed commit can contribute, which is why this is derived
        from `committed` rather than from the number of documents offered.
        """
        return self.pages_written if self.committed else 0


class IndexerAgent:
    """Fail-closed persistence."""

    def __init__(self, store: IndexStore, guard: StorageGuard) -> None:
        self._store = store
        self._guard = guard

    async def storage_state(self) -> StorageState:
        return await self._guard.storage_state()

    async def persist(
        self,
        documents: Sequence[PageDocument],
        queue_ids: Sequence[int] = (),
    ) -> IndexOutcome:
        """Write a batch, or explain why it was not written."""
        try:
            await self._guard.assert_can_write()
        except StoragePaused as error:
            logger.warning("indexer paused: %s", error)
            return IndexOutcome(committed=False, reason=str(error), paused=True)

        outcome: PersistOutcome = await self._store.persist_batch(documents, queue_ids)
        if not outcome.committed:
            return IndexOutcome(committed=False, reason=outcome.reason)

        # The guard's cached count is now stale by exactly what we just wrote.
        self._guard.note_write(outcome.pages_written)

        return IndexOutcome(
            committed=True,
            pages_written=outcome.pages_written,
            duplicates=outcome.duplicates,
            queue_closed=outcome.queue_closed,
            total_pages=outcome.total_pages,
        )

    async def mark_failed(self, queue_ids: Sequence[int], reason: str = "") -> int:
        return await self._store.mark_failed(queue_ids, reason)

    async def mark_skipped(self, queue_ids: Sequence[int], reason: str = "") -> int:
        return await self._store.mark_skipped(queue_ids, reason)


__all__ = ["IndexOutcome", "IndexerAgent"]

"""Agent 06 — revalidation.

The index goes stale: timings change, prices move, a page is replaced by a
holding page. This worker walks the oldest rows, re-fetches them, and updates
the row in place when the fresh copy still passes the quality gate.

It runs on a long, jittered interval so a fleet of instances never revalidates
in lockstep, and it never raises out of its loop — a dead host is logged and
skipped, and the pass carries on.
"""

from __future__ import annotations

import asyncio
import logging
import random

from app.agents import load_agent
from app.engine.config import Settings, get_settings
from app.engine.store import IndexStore, PageDocument

logger = logging.getLogger(__name__)

#: The worker never sleeps less than this, however it is configured.
MIN_SLEEP_SECONDS = 30.0

_parser_module = load_agent("03_parser_agent")
_quality_module = load_agent("04_quality_agent")

ParserAgent = _parser_module.ParserAgent
QualityAgent = _quality_module.QualityAgent


class RevalidationAgent:
    """Re-fetches the stalest rows and refreshes them in place."""

    def __init__(
        self,
        store: IndexStore,
        fetcher,
        settings: Settings | None = None,
        parser: object | None = None,
        quality: object | None = None,
    ) -> None:
        self._store = store
        self._fetcher = fetcher
        self._settings = settings or get_settings()
        self._parser = parser or ParserAgent()
        self._quality = quality or QualityAgent()
        self.refreshed = 0
        self.failed = 0

    async def run_once(self) -> int:
        """One revalidation pass. Returns how many rows were refreshed."""
        stale = await self._store.stale_pages(
            limit=self._settings.revalidation_batch_size,
            older_than_seconds=self._settings.revalidation_stale_after_seconds,
        )
        if not stale:
            return 0

        self._quality.reset()
        refreshed = 0
        for url in stale:
            try:
                result = await self._fetcher.fetch(url, attempts=1)
                if not result.ok:
                    self.failed += 1
                    logger.debug("revalidation skipped %s: %s", url, result.reason)
                    continue

                parsed = self._parser.parse(url, result.html)
                if not parsed.ok:
                    self.failed += 1
                    continue

                domain = self._domain(url)
                document = PageDocument(
                    url=url,
                    domain=domain,
                    title=parsed.title,
                    description=parsed.description,
                    body_text=parsed.body_text,
                )
                verdict = self._quality.check(document)
                if not verdict.ok:
                    self.failed += 1
                    logger.debug("revalidation rejected %s: %s", url, verdict.reason)
                    continue

                document.digest = verdict.digest
                document.quality_score = verdict.score
                if await self._store.update_page(document):
                    refreshed += 1
            except Exception as error:  # never let one URL end the pass
                self.failed += 1
                logger.warning("revalidation error for %s: %s", url, error)

        self.refreshed += refreshed
        if refreshed:
            logger.info("revalidation: refreshed %d of %d stale pages", refreshed, len(stale))
        return refreshed

    async def run_forever(self) -> None:
        """Loop until cancelled, sleeping a jittered, never-short interval."""
        while True:
            try:
                await self.run_once()
            except asyncio.CancelledError:
                raise
            except Exception as error:  # a broken pass must not kill the worker
                logger.error("revalidation pass failed: %s", error)

            jitter = random.uniform(0, max(0, self._settings.revalidation_jitter_seconds))
            delay = max(
                MIN_SLEEP_SECONDS,
                self._settings.revalidation_interval_seconds + jitter,
            )
            logger.debug("revalidation worker sleeping %.1fs", delay)
            await asyncio.sleep(delay)

    @staticmethod
    def _domain(url: str) -> str:
        from urllib.parse import urlparse

        host = (urlparse(url).hostname or "").lower()
        return host[4:] if host.startswith("www.") else host


__all__ = ["MIN_SLEEP_SECONDS", "RevalidationAgent"]

"""Agent 08 — the language agent.

Detects the script a page is written in, so a search can be filtered by
language. Three outcomes: `kn` (Kannada), `hi` (Devanagari) and `en` (Latin,
the fallback).

Detection is by Unicode block, not by word lists: a page in Kannada is written
in the Kannada block whatever its vocabulary, and no dictionary is needed. A
script has to carry a real share of the text before it wins, so an English page
with a Kannada menu does not get labelled Kannada.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass

logger = logging.getLogger(__name__)

#: Unicode blocks, inclusive.
SCRIPT_RANGES: dict[str, tuple[int, int]] = {
    "kn": (0x0C80, 0x0CFF),
    "hi": (0x0900, 0x097F),
}

DEFAULT_LANGUAGE = "en"
SUPPORTED_LANGUAGES: tuple[str, ...] = ("en", "hi", "kn")

#: A script must be at least this share of the letters to be the language.
MIN_SCRIPT_SHARE = 0.15

#: ...and at least this many characters, so a stray glyph does not decide it.
MIN_SCRIPT_CHARS = 3


@dataclass
class Detection:
    """The detected language and the evidence for it."""

    language: str = DEFAULT_LANGUAGE
    counts: dict[str, int] | None = None
    share: float = 0.0

    @property
    def is_fallback(self) -> bool:
        return self.language == DEFAULT_LANGUAGE and self.share == 0.0


class LanguageAgent:
    """Detects `kn`, `hi` or `en` from the characters actually present."""

    def detect(self, text: str) -> str:
        return self.detect_detail(text).language

    def detect_detail(self, text: str) -> Detection:
        if not text:
            return Detection(language=DEFAULT_LANGUAGE, counts={}, share=0.0)

        counts = {language: 0 for language in SCRIPT_RANGES}
        letters = 0

        for character in text:
            if not character.isalpha():
                continue
            letters += 1
            codepoint = ord(character)
            for language, (low, high) in SCRIPT_RANGES.items():
                if low <= codepoint <= high:
                    counts[language] += 1
                    break

        if not letters:
            return Detection(language=DEFAULT_LANGUAGE, counts=counts, share=0.0)

        language, count = max(counts.items(), key=lambda item: item[1])
        share = count / letters

        if count >= MIN_SCRIPT_CHARS and share >= MIN_SCRIPT_SHARE:
            return Detection(language=language, counts=counts, share=round(share, 4))

        return Detection(language=DEFAULT_LANGUAGE, counts=counts, share=round(share, 4))


__all__ = [
    "DEFAULT_LANGUAGE",
    "Detection",
    "LanguageAgent",
    "MIN_SCRIPT_CHARS",
    "MIN_SCRIPT_SHARE",
    "SCRIPT_RANGES",
    "SUPPORTED_LANGUAGES",
]

"""Agent 07 — the classifier.

Tags every page at ingestion time so a search can be narrowed without a
backfill. Ten categories, scored from three signals in descending order of
trust: the domain's public suffix, tokens in the URL, then words in the title,
description and body.

Suffixes are weighted highest because they are structural — `.gov.in` is a
government domain whatever its copy says. Body keywords are weighted lowest

import re
from dataclasses import dataclass, field
from urllib.parse import urlparse

)

#: Scores. A suffix outranks a single URL token, so a bare `.gov.in` page is
#: government whatever its copy says — but a page carrying an explicit topical

SUFFIX_WEIGHT = 4
URL_TOKEN_WEIGHT = 3

@dataclass
class Classification:
    """The chosen category and why it won."""

    category: str = DEFAULT_CATEGORY
    score: int = 0
    matched: list[str] = field(default_factory=list)

    @property
    def is_fallback(self) -> bool:
        return self.category == DEFAULT_CATEGORY and not self.matched


#: Split a URL into words. Substring matching is wrong here: "example.com"
#: contains "exam" and would be classified as an exam page.
WORD_SPLIT = re.compile(r"[^a-z0-9]+")


def _url_words(host: str, path: str) -> set[str]:
    return {word for word in WORD_SPLIT.split(f"{host} {path}") if word}


def _token_matches(token: str, words: set[str]) -> bool:
    """A multi-word token like `admit-card` needs all of its parts present."""
    parts = [part for part in token.split("-") if part]
    return bool(parts) and all(part in words for part in parts)


def _host_and_path(url: str) -> tuple[str, str]:
    parts = urlparse(url)
    host = (parts.hostname or "").lower()
    if host.startswith("www."):
        host = host[4:]
    return host, (parts.path or "").lower()


class ClassifierAgent:
    """Assigns one of ten categories to a page."""

    def __init__(self, rules: tuple[CategoryRule, ...] = RULES) -> None:
        self._rules = rules

    def classify(
        self,
        url: str,
        title: str = "",
        description: str = "",
        body_text: str = "",
    ) -> Classification:
        host, path = _host_and_path(url)
        headline = f"{title} {description}".lower()
        body = (body_text or "").lower()

        best = Classification()
        for rule in self._rules:
            score, matched = self._score(rule, host, path, headline, body)
            if score > best.score:
                best = Classification(category=rule.name, score=score, matched=matched)

        return best

    @staticmethod
    def _score(
        rule: CategoryRule,
        host: str,
        path: str,
        headline: str,
        body: str,
    ) -> tuple[int, list[str]]:
        score = 0
        matched: list[str] = []

        for suffix in rule.suffixes:
            if host.endswith(suffix):
                score += SUFFIX_WEIGHT
                matched.append(f"suffix:{suffix}")
                break

        words = _url_words(host, path)
        for token in rule.url_tokens:
            if _token_matches(token, words):
                score += URL_TOKEN_WEIGHT
                matched.append(f"url:{token}")

        for keyword in rule.keywords:
            if keyword in headline:
                score += TEXT_TOKEN_WEIGHT
                matched.append(f"text:{keyword}")

        body_hits = 0
        for keyword in rule.keywords:
            if body_hits >= MAX_BODY_HITS:
                break
            if keyword in body:
                score += BODY_KEYWORD_WEIGHT
                body_hits += 1
                matched.append(f"body:{keyword}")

        return score, matched

"""The index store: every read and write of the page index.

Two rules shape this module.

**Persistence is one trip and it is atomic.** A page write and the matching
`crawl_queue` transition go into a single `BEGIN IMMEDIATE … COMMIT` batch. If
any statement in that batch fails, none of it lands and the URL is not marked
processed — which is the difference between "the crawl made progress" and "the
queue drained into nothing".

**Counts are proved, not assumed.** `persist_batch` reads `COUNT(*) FROM pages`
inside the same transaction that wrote the rows, so the number it reports is a
number the database confirmed, not a number the caller hoped for.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass, field
from typing import Any, Iterable, Sequence

from app.engine.config import Settings, get_settings
from app.libsql_http import LibsqlClient, LibsqlError

logger = logging.getLogger(__name__)

# Columns the FTS index carries, in bm25 weight order.
BM25_WEIGHTS = (10.0, 5.0, 1.0)

#: How much a title match is worth on top of bm25. bm25 is negative and
#: lower-is-better, so this is subtracted from the score to promote the row.
TITLE_MATCH_BOOST = 1.0

#: How many query terms are checked against the title.
MAX_TITLE_TOKENS = 5

#: Width of the passage returned as a match snippet, in characters.
SNIPPET_CHAR_CAP = 320


@dataclass
class PageDocument:
    """One page, ready to be stored."""

    url: str
    domain: str
    title: str
    description: str
    body_text: str
    quality_score: float = 1.0
    digest: str = ""
    category: str = "general_reference"
    language: str = "en"

    def as_row(self) -> tuple[Any, ...]:
        """The nine bound values the page insert expects, in SQL order.

        `quality_score` is cast to `float` here rather than trusted from the
        caller: it is a REAL column, and an `int` reaching the encoder would be
        sent as an integer while a `Decimal` would be sent as text. Turso
        rejects both for a REAL column.

        return (
            self.domain,
            self.url,
            self.title,
            self.description,
            self.body_text,
            float(self.quality_score),
            self.digest,
            self.category,
            self.language,
        )

    description: str
    body_text: str

    @property
    def snippet(self) -> str:
        """The passage that matched the query, else the description.

        `snippet()` from FTS5 picks the part of the page the query actually
        hit, which beats a fixed prefix of the body. When the match was on the
        title the snippet is the title again, so it falls back rather than
        repeating the heading back at the reader.

        matched = " ".join((self.match_snippet or "").split())
        title = " ".join((self.title or "").split())
        if matched and matched.strip("…").strip() != title:
            return matched[:SNIPPET_CHAR_CAP]

        return body[:200] + ("…" if len(body) > 200 else "")

_FTS_UNSAFE = set('"\':*()[]{}^~+-')


def fts_tokens(query: str) -> list[str]:
    """Split a user query into FTS-safe terms.

    FTS5 treats `-`, `:`, `*`, quotes and brackets as syntax, so a query of
    "covid-19: what now?" is a syntax error unless every term is quoted. Each
    token is therefore reduced to word characters and emitted as a quoted
    string.
    """
    tokens: list[str] = []
    for raw in (query or "").lower().split():
        cleaned = "".join(ch for ch in raw if ch.isalnum() or ch in "_/.")
        cleaned = cleaned.strip("_/.")
        if cleaned:
            tokens.append(cleaned)
    return tokens


def build_match_query(query: str, *, prefix: bool = False) -> str:
    """Turn a user query into an FTS5 MATCH expression."""
    tokens = fts_tokens(query)
    if not tokens:
        return ""
    if prefix:
        return " OR ".join(f'"{token}"*' for token in tokens)
    return " AND ".join(f'"{token}"' for token in tokens)


def passage_around_match(text: str, tokens: Sequence[str], window: int = SNIPPET_CHAR_CAP) -> str:
    """The window of `text` around the first query term it contains.

    FTS5's own `snippet()` was the obvious tool and is what the brief asked
    for, but on the SQLite build this project runs on it ignores `colnum` and
    returns column 0 for every call — verified against both a plain and an
    external-content FTS5 table, where `highlight()` behaves correctly and
    `snippet()` does not. Rather than depend on behaviour that is wrong here
    and unverifiable on Turso, the window is computed in Python: portable,
    deterministic and testable.

    Returns `""` when no token appears, so the caller can fall back to the
    description instead of showing an arbitrary opening.
    """
    cleaned = " ".join((text or "").split())
    if not cleaned or not tokens:
        return ""

    lowered = cleaned.lower()
    positions = [lowered.find(token) for token in tokens if token]
    positions = [position for position in positions if position >= 0]
    if not positions:
        return ""

    match_at = min(positions)
    start = max(0, match_at - window // 3)
    end = min(len(cleaned), start + window)
    start = max(0, min(start, end - window))

    # Do not cut a word in half at either edge.
    if start > 0:
        space = cleaned.find(" ", start)
        if 0 <= space < match_at:
            start = space + 1
    if end < len(cleaned):
        space = cleaned.rfind(" ", match_at, end)
        if space > match_at:
            end = space

    passage = cleaned[start:end].strip()
    if start > 0:
        passage = "…" + passage
    if end < len(cleaned):
        passage = passage + "…"
    return passage


def normalise_score(raw: float) -> float:
    """bm25 is lower-is-better and unbounded; map it to 0..1 for the API."""
    try:
        value = abs(float(raw))
    except (TypeError, ValueError):
        return 0.0
    return round(1.0 / (1.0 + value), 4)

class IndexStore:
    """All index reads and writes, over one injected database client."""

    def __init__(self, client: LibsqlClient, settings: Settings | None = None) -> None:
        self._client = client
        self._settings = settings or get_settings()

    @property
    def client(self) -> LibsqlClient:
        return self._client

    async def ensure_schema(self) -> dict[str, int]:
        from app.database import initialize_database

        return await initialize_database(self._client)

    # -- writes ------------------------------------------------------------

    async def persist_batch(
        self,
        documents: Sequence[PageDocument],
        queue_ids: Sequence[int] = (),
    ) -> PersistOutcome:
        """Write pages and close their queue rows in one atomic batch."""
        if not documents and not queue_ids:
            return PersistOutcome(committed=True, reason="nothing to write")

        statements: list[tuple[str, Sequence[Any]]] = [("BEGIN IMMEDIATE", ())]

        for document in documents:
            statements.append(
                (
                    "INSERT INTO sites (domain, canonical_url) VALUES (?, ?) "
                    "ON CONFLICT(domain) DO NOTHING",
                    (document.domain, f"https://{document.domain}/"),
                )
            )

        for document in documents:
            statements.append(
                (
                    "INSERT OR IGNORE INTO pages "
                    "(site_id, url, title, description, body_text, quality_score, digest, "
                    "category, language, indexed_at) "
                    "VALUES ((SELECT site_id FROM sites WHERE domain = ?), ?, ?, ?, ?, ?, ?, "
                    "?, ?, CURRENT_TIMESTAMP)",
                    document.as_row(),
                )
            )

        for rowid in queue_ids:
            statements.append(
                (
                    "UPDATE crawl_queue SET status = 'processed', "
                    "processed_at = CURRENT_TIMESTAMP WHERE rowid = ?",
                    (rowid,),
                )
            )

        statements.append(("SELECT COUNT(*) FROM pages", ()))
        statements.append(("SELECT COUNT(*) FROM sites", ()))
        statements.append(("COMMIT", ()))

        try:
            results = await self._client.batch(statements)
        except LibsqlError as error:
            await self._rollback_quietly()
            logger.error("persistence batch failed: %s", error)
            return PersistOutcome(committed=False, reason=str(error))

        pages_start = 1 + len(documents)
        queue_start = pages_start + len(documents)
        page_inserts = results[pages_start:queue_start]
        queue_updates = results[queue_start : queue_start + len(queue_ids)]
        total_pages = int(results[-3].scalar(0) or 0)
        total_sites = int(results[-2].scalar(0) or 0)

        written = sum(1 for result in page_inserts if result.rows_affected > 0)
        closed = sum(1 for result in queue_updates if result.rows_affected > 0)

        outcome = PersistOutcome(
            pages_written=written,
            total_pages=total_pages,
            sites_written=total_sites,
            queue_closed=closed,
            duplicates=max(0, len(documents) - written),
            committed=True,
        )
        logger.info(
            "DB COMMIT SUCCESS: %d pages written, %d duplicates skipped, %d queue rows closed "
            "(index total: %d pages, %d sites)",
            outcome.pages_written,
            outcome.duplicates,
            outcome.queue_closed,
            outcome.total_pages,
            outcome.sites_written,
        )
        return outcome

    async def domain_page_counts(self, domains: Sequence[str]) -> dict[str, int]:
        """Indexed pages per domain, for the whole batch in one query.

        `pages` has no `domain` column — it is reached through `sites` — and the
        count is deliberately batched: one query per batch, never one per URL,
        because a per-URL count would put a database round trip in front of
        every fetch.
        """
        unique = [domain for domain in dict.fromkeys(domains) if domain]
        if not unique:
            return {}
        placeholders = ", ".join("?" for _ in unique)
        result = await self._client.execute(
            "SELECT s.domain, COUNT(*) FROM pages p "
            "JOIN sites s ON s.site_id = p.site_id "
            f"WHERE s.domain IN ({placeholders}) GROUP BY s.domain",
            tuple(unique),
        )
        return {str(row[0]): int(row[1]) for row in result.rows}

    async def mark_skipped(self, queue_ids: Sequence[int], reason: str = "") -> int:
        """Park queue rows that are deliberately not crawled.

        Distinct from `failed`: a skipped row is a policy decision (the domain
        is at its page cap), not an error, and re-arming never revives it.

        if not queue_ids:

        results = await self._client.batch(

        skipped = sum(1 for result in results if result.rows_affected > 0)

        return skipped

    async def mark_failed(self, queue_ids: Sequence[int], reason: str = "") -> int:

        if not queue_ids:

    async def enqueue_urls(self, urls: Iterable[tuple[str, str]]) -> int:

    async def rearm_processed(self, limit: int = 25, cooldown_seconds: int = 6 * 3600) -> int:
        """Return the stalest crawled URLs to `pending` once their cooldown ends.

        Without this the frontier starves: once every seed is in `pages`, no

        the crawl alive and the index fresh, bounded by a cooldown so a URL is

        result = await self._client.execute(
            "UPDATE crawl_queue SET status = 'pending', processed_at = NULL "
            "WHERE rowid IN ("

            "  WHERE status = 'processed'"
            "    AND (processed_at IS NULL OR processed_at <= datetime('now', ?))"
            "  ORDER BY COALESCE(processed_at, '1970-01-01 00:00:00') ASC"
            "  LIMIT ?"

        return result.rows_affected

    # -- search ------------------------------------------------------------

    async def search(
        self,
        query: str,
        limit: int = 10,
        offset: int = 0,
        category: str | None = None,
        language: str | None = None,
    ) -> list[SearchHit]:
        """Weighted FTS5 search, best match first, optionally filtered."""
        match = build_match_query(query)
        if not match:
            return []

        weights = ", ".join(str(weight) for weight in BM25_WEIGHTS)

        filters = ""
        extra: list[Any] = []
        if category:
            filters += " AND p.category = ?"
            extra.append(category)
        if language:
            filters += " AND p.language = ?"
            extra.append(language)

        # Title boost. Bound parameters appear in SQL order, so the title
        # terms (in the SELECT) are bound before the MATCH term (in the WHERE).
        tokens = fts_tokens(query)
        title_tokens = tokens[:MAX_TITLE_TOKENS]
        if title_tokens:
            title_terms = " OR ".join("instr(lower(p.title), ?) > 0" for _ in title_tokens)
        else:
            title_terms = "0"

        sql = (
            "SELECT p.page_id, p.url, p.title, p.description, p.body_text, "
            f"bm25(page_fts, {weights}) AS lexical_score, "
            f"CASE WHEN ({title_terms}) THEN 1 ELSE 0 END AS title_hit "
            "FROM page_fts JOIN pages p ON p.page_id = page_fts.rowid "
            f"WHERE page_fts MATCH ?{filters} "
            "ORDER BY (lexical_score - ? * title_hit) ASC LIMIT ? OFFSET ?"
        )
        args = (*title_tokens, match, *extra, TITLE_MATCH_BOOST, limit, offset)

        result = await self._client.execute(sql, args)
        hits = [self._row_to_hit(row, tokens) for row in result.rows]

        if hits:
            self._rescale_scores(hits)
            return hits

        # A strict AND is right for precision and wrong for a half-remembered
        # query. Fall back to OR rather than answering "nothing found".
        loose = build_match_query(query, prefix=True)
        if loose and loose != match:
            result = await self._client.execute(
                sql, (*title_tokens, loose, *extra, TITLE_MATCH_BOOST, limit, offset)
            )
            hits = [self._row_to_hit(row, tokens) for row in result.rows]
            self._rescale_scores(hits)
        return hits

    @staticmethod
    def _row_to_hit(row: Sequence[Any], tokens: Sequence[str] = ()) -> SearchHit:
        title_hit = int(row[6] or 0)
        boosted = float(row[5] or 0.0) - TITLE_MATCH_BOOST * title_hit
        body = str(row[4] or "")
        return SearchHit(
            page_id=int(row[0]),
            url=str(row[1] or ""),
            title=str(row[2] or ""),
            description=str(row[3] or ""),
            body_text=body,
            score=normalise_score(boosted),
            match_snippet=passage_around_match(body, tokens),
            raw_score=boosted,
        )

    @staticmethod
    def _rescale_scores(hits: Sequence[SearchHit]) -> None:
        """Spread `score` across the page of results, best first.

        bm25 is negative and lower-is-better, and the title boost pushes a
        good row below zero — so mapping it absolutely would make a boosted
        result look *worse* than an unboosted one. Rescaling against the page
        keeps the number monotone with the order the caller actually receives.
        It is relative within one page of results, not comparable across pages.
        """
        if not hits:
            return
        best = min(hit.raw_score for hit in hits)
        worst = max(hit.raw_score for hit in hits)
        span = worst - best
        for hit in hits:
            hit.score = 1.0 if span <= 0 else round((worst - hit.raw_score) / span, 4)

    async def suggest(self, query: str, limit: int = 6) -> list[str]:
        """Title completions for the search bar."""
        match = build_match_query(query, prefix=True)
        if not match:
            return []
        result = await self._client.execute(
            "SELECT p.title FROM page_fts JOIN pages p ON p.page_id = page_fts.rowid "
            "WHERE page_fts MATCH ? AND p.title <> '' ORDER BY rank LIMIT ?",
            (match, limit),
        )
        seen: set[str] = set()
        titles: list[str] = []
        for row in result.rows:
            title = " ".join(str(row[0] or "").split())
            if title and title.lower() not in seen:
                seen.add(title.lower())
                titles.append(title)
        return titles

    async def stats(self) -> Stats:
        results = await self._client.batch(
            [
                ("SELECT COUNT(*) FROM sites", ()),
                ("SELECT COUNT(*) FROM pages", ()),
                ("SELECT COUNT(*) FROM page_fts", ()),
                ("SELECT COUNT(*) FROM crawl_queue WHERE status = 'pending'", ()),
            ]
        )
        return Stats(
            sites=int(results[0].scalar(0) or 0),
            pages=int(results[1].scalar(0) or 0),
            indexed=int(results[2].scalar(0) or 0),
            queued=int(results[3].scalar(0) or 0),
        )

    # -- revalidation ------------------------------------------------------

    async def stale_pages(self, limit: int = 5, older_than_seconds: int = 7 * 24 * 3600) -> list[str]:
        result = await self._client.execute(

    async def get_page(self, url: str) -> dict[str, Any] | None:

        return {

            "title": str(row[2] or ""),
            "description": str(row[3] or ""),

        }

"""Storage safety: stop writing before the database bill does.

`StorageGuard` is the only thing in the process allowed to answer "how many
pages do we hold". Two rules make it safe:

* **A 60-second in-memory TTL cache.** The crawl pipeline asks once per batch,
  never once per URL. `COUNT(*)` on every fetch is the difference between a
  free tier that holds and one that rate-limits itself into a stall.
* **Fail-closed.** If the count cannot be read — the database is down, the
  token expired, the response was malformed — the guard reports `paused=True`
  and blocks writes. Crawling blind past a ceiling is how a 300,000-page budget
  becomes a bill; refusing to write until the count is known costs nothing but
  a delay.
"""

from __future__ import annotations

import asyncio
import logging
import time
from dataclasses import dataclass

from app.libsql_http import LibsqlClient, LibsqlError

logger = logging.getLogger(__name__)

STORAGE_WARNING = (
    "WARNING: Storage safety threshold reached (300k+ records). "
    "Pausing autonomous crawling."
)


@dataclass(frozen=True)
class StorageState:
    """What the guard knows right now, and whether writes are allowed."""

    pages: int
    sites: int
    limit: int
    paused: bool
    reason: str = ""
    from_cache: bool = False

    @property
    def over_limit(self) -> bool:
        return self.pages >= self.limit

    def as_dict(self) -> dict[str, object]:
        return {
            "pages": self.pages,
            "sites": self.sites,
            "limit": self.limit,
            "over_limit": self.over_limit,
            "paused": self.paused,
            "reason": self.reason,
        }


class StorageGuard:
    """Counts pages and sites at most once per TTL, and fails closed."""

    def __init__(
        self,
        client: LibsqlClient,
        *,
        limit: int = 300_000,
        ttl_seconds: float = 60.0,
        clock=time.monotonic,
    ) -> None:
        self._client = client
        self._limit = limit
        self._ttl = ttl_seconds
        self._clock = clock
        self._lock = asyncio.Lock()
        self._cached: StorageState | None = None
        self._cached_at = 0.0
        self._queries = 0

    @property
    def limit(self) -> int:
        return self._limit

    @property
    def count_queries(self) -> int:
        """How many times the database was actually asked. Tests assert on this."""
        return self._queries

    def invalidate(self) -> None:
        """Force the next call to read the database again."""
        self._cached = None
        self._cached_at = 0.0

    def note_write(self, pages: int) -> None:
        """Add pages we just committed to the cached count.

        The pipeline knows exactly how many rows it wrote, so it can keep the
        cache honest between refreshes without spending a query.
        """
        if self._cached is None or pages <= 0:
            return
        self._cached = StorageState(
            pages=self._cached.pages + pages,
            sites=self._cached.sites,
            limit=self._cached.limit,
            paused=self._cached.paused,
            reason=self._cached.reason,
            from_cache=True,
        )

    async def storage_state(self) -> StorageState:
        """The current state, from cache when fresh, fail-closed when not."""
        async with self._lock:
            now = self._clock()
            if self._cached is not None and (now - self._cached_at) < self._ttl:
                return StorageState(
                    pages=self._cached.pages,
                    sites=self._cached.sites,
                    limit=self._cached.limit,
                    paused=self._cached.paused,
                    reason=self._cached.reason,
                    from_cache=True,
                )

            try:
                results = await self._client.batch(
                    [
                        ("SELECT COUNT(*) FROM pages", ()),
                        ("SELECT COUNT(*) FROM sites", ()),
                    ]
                )
                self._queries += 1
                pages = int(results[0].scalar(0) or 0)
                sites = int(results[1].scalar(0) or 0)
            except (LibsqlError, ValueError, TypeError, IndexError) as error:
                # Fail closed: an unknown page count is not permission to write.
                state = StorageState(
                    pages=self._limit,
                    sites=0,
                    limit=self._limit,
                    paused=True,
                    reason=f"storage_state unavailable: {error}",
                    from_cache=False,
                )
                self._cached = state
                self._cached_at = now
                logger.error("storage guard failed closed: %s", error)
                return state

            over = pages >= self._limit
            state = StorageState(
                pages=pages,
                sites=sites,
                limit=self._limit,
                paused=over,
                reason=STORAGE_WARNING if over else "",
                from_cache=False,
            )
            self._cached = state
            self._cached_at = now
            if over:
                logger.warning(STORAGE_WARNING)
            return state

    async def can_write(self) -> bool:
        state = await self.storage_state()
        return not state.paused

    async def assert_can_write(self) -> StorageState:
        """Raise when writes are blocked, so a caller cannot ignore it."""
        state = await self.storage_state()
        if state.paused:
            raise StoragePaused(state.reason or "storage paused")
        return state


class StoragePaused(RuntimeError):
    """Writes are blocked because the index is at its ceiling or unreadable."""


__all__ = ["STORAGE_WARNING", "StorageGuard", "StoragePaused", "StorageState"]

"""Background worker lifecycle.

The workers are started from the FastAPI lifespan with `asyncio.create_task`,
which returns as soon as the task is scheduled. The port binds and `/health`
answers immediately; the crawl happens behind it.

Stopping is symmetric: every task is cancelled and awaited, so a reload or a
Render redeploy never leaves a task writing to a closed client.
"""

from __future__ import annotations

import asyncio
import logging
from dataclasses import dataclass, field

from app.engine.config import Settings, get_settings
from app.orchestrator import Orchestrator

logger = logging.getLogger(__name__)


@dataclass
class WorkerHandles:
    """The running background tasks, so the lifespan can stop them."""

    tasks: dict[str, asyncio.Task] = field(default_factory=dict)

    def add(self, name: str, task: asyncio.Task) -> None:
        self.tasks[name] = task

    @property
    def names(self) -> list[str]:
        return sorted(self.tasks)

    def running(self) -> list[str]:
        return [name for name, task in self.tasks.items() if not task.done()]


async def _guarded(name: str, coro) -> None:
    """Run a worker loop and swallow only its final exception, loudly."""
    try:
        await coro
    except asyncio.CancelledError:
        logger.info("worker stopped: %s", name)
        raise
    except Exception as error:  # a dead worker must be visible, not silent
        logger.exception("worker %s died: %s", name, error)


def start_workers(
    orchestrator: Orchestrator,
    settings: Settings | None = None,
) -> WorkerHandles:
    """Schedule the crawl and revalidation loops without blocking startup."""
    settings = settings or get_settings()
    handles = WorkerHandles()

    if settings.crawler_enabled:
        handles.add(
            "crawler",
            asyncio.create_task(_guarded("crawler", orchestrator.run_forever())),
        )
        logger.info("crawl worker started")

    if settings.revalidation_enabled:
        handles.add(
            "revalidation",
            asyncio.create_task(
                _guarded("revalidation", orchestrator.revalidation.run_forever())
            ),
        )
        logger.info("revalidation worker started")

    return handles


async def stop_workers(handles: WorkerHandles | None) -> None:
    """Cancel every worker and wait for it to unwind."""
    if handles is None:
        return
    for name, task in handles.tasks.items():
        if not task.done():
            task.cancel()
    for name, task in handles.tasks.items():
        try:
            await task
        except asyncio.CancelledError:
            pass
        except Exception as error:  # already logged by the guard
            logger.debug("worker %s raised while stopping: %s", name, error)
    logger.info("all workers stopped")


__all__ = ["WorkerHandles", "start_workers", "stop_workers"]

def youtube_thumbnail(url: str) -> str | None:
    """The deterministic thumbnail for a YouTube video URL, or `None`.

    `i.ytimg.com/vi/<id>/hqdefault.jpg` is a pure function of the video id, so
    this is a derived value rather than a guess.
    """
    host = (urlparse(url).hostname or "").lower()
    if host.startswith("www."):
        host = host[4:]
    if host not in _YOUTUBE_HOSTS:
        return None
    match = _YOUTUBE_ID.search(url)
    if not match:
        return None
    return f"https://i.ytimg.com/vi/{match.group(1)}/hqdefault.jpg"


def kind_of(url: str) -> str:
    """A coarse content type, derived only from the URL itself."""
    path = (urlparse(url).path or "").lower()
    host = (urlparse(url).hostname or "").lower()
    if host.endswith(_YOUTUBE_HOSTS) or host == "youtu.be":
        return "video"
    if path.endswith(_DOCUMENT_EXTENSIONS):
        return "document"
    if path.endswith(_IMAGE_EXTENSIONS):
        return "image"
    if any(host.endswith(news_host) for news_host in _NEWS_HOSTS):
        return "news"
    return "web"

class SearchService:
    """Search, suggest and retrieve over the index."""

    def __init__(self, store: IndexStore, settings: Settings | None = None) -> None:
        self._store = store
        self._settings = settings or get_settings()

    async def search(
        self,
        query: str,
        limit: int | None = None,
        page: int = 1,
        category: str | None = None,
        language: str | None = None,
    ) -> dict[str, Any]:
        """The `GET /v1/search` payload."""
        started = time.perf_counter()
        size = self._clamp_limit(limit)
        page_number = max(1, min(page, self._settings.search_max_page))
        offset = (page_number - 1) * size

        hits = await self._store.search(
            query,
            limit=size,
            offset=offset,
            category=category,
            language=language,
        )
        results = [to_result(hit).as_dict() for hit in hits]
        took_ms = int((time.perf_counter() - started) * 1000)

        return {
            "query": query,
            "page": page_number,
            "limit": size,
            "total_estimate": len(results),
            "took_ms": took_ms,
            "degraded": False,
            "cache_hit": True,
            "providers": {SOURCE_LABEL: len(results)},
            "failed_providers": [],
            "filters": {"category": category, "language": language},
            "results": results,
        }

    async def search_flat(
        self,
        query: str,
        limit: int | None = None,
        category: str | None = None,
        language: str | None = None,
    ) -> dict[str, Any]:
        """The meta-search contract (`format=json`), shaped like the README."""
        started = time.perf_counter()
        size = self._clamp_limit(limit)
        hits = await self._store.search(
            query, limit=size, category=category, language=language
        )
        return {
            "query": query,
            "total_hits": len(hits),
            "took_ms": int((time.perf_counter() - started) * 1000),
            "source": SOURCE_LABEL,
            "results": [
                {
                    "title": hit.title or hit.url,
                    "url": hit.url,
                    "snippet": hit.snippet,
                    "favicon": self._favicon(hit.url),
                }
                for hit in hits
            ],
        }

    async def suggest(self, query: str, limit: int = 6) -> list[Any]:
        """The OpenSearch suggestions shape: `[query, [completions]]`."""
        try:
            titles = await self._store.suggest(query, limit=limit)
        except Exception as error:  # suggestions are a convenience, never an error
            logger.debug("suggest failed for %r: %s", query, error)
            return [query, []]
        return [query, titles]

    async def retrieve(
        self,
        query: str,
        limit: int | None = None,
        format: str = "json",
    ) -> dict[str, Any] | str:
        """Token-optimised context for AI clients."""
        size = self._clamp_limit(limit)
        hits = await self._store.search(query, limit=size)
        results = [
            {"title": hit.title or hit.url, "url": hit.url, "snippet": hit.snippet}
            for hit in hits
        ]
        if format == "markdown":
            return self._as_markdown(query, results)
        return {
            "query": query,
            "format": format,
            "degraded": False,
            "providers": {SOURCE_LABEL: len(results)},
            "failed_providers": [],
            "results": results,
            "context": [
                {"title": hit.title or hit.url, "url": hit.url, "text": hit.snippet}
                for hit in hits
            ],
        }

    async def stats(self) -> dict[str, Any]:
        stats = await self._store.stats()
        return stats.as_dict()

    @staticmethod
    def normalise_category(category: str | None) -> str | None:
        """Validate a `category` filter, or `None` when it was not supplied."""
        if category is None or not category.strip():
            return None
        value = category.strip().lower()
        if value not in CATEGORIES:
            raise ValueError(f"unknown category '{category}'")
        return value

    @staticmethod
    def normalise_language(language: str | None) -> str | None:
        """Validate a `lang` filter, or `None` when it was not supplied."""
        if language is None or not language.strip():
            return None
        value = language.strip().lower()
        if value not in SUPPORTED_LANGUAGES:
            raise ValueError(f"unsupported language '{language}'")
        return value

    def _clamp_limit(self, limit: int | None) -> int:

"""The one path to the database.

Two things live here:

* `ProcessRateLimiter` — a single process-wide throttle so the crawler, the
  revalidation worker and the search path together never exceed two Turso
  requests per second. Turso bills and throttles per request; without one
  shared limiter each worker keeps its own idea of "politeness" and the sum of
  them is a request storm.
* `TursoHttpClient` — a small client for Turso's `/v2/pipeline` endpoint that
  checks every step. Turso answers HTTP 200 with a per-step error object, so a
  client that only looks at the status code reports success for a write that
  never happened.

A `LocalSqliteClient` with the same interface backs `file:` URLs and the test
suite, so the whole application can run without a network.
"""

# A statement plus its bound arguments, in the shape both clients accept.
Statement = tuple[str, Sequence[Any]]


class LibsqlError(RuntimeError):
    """A statement the database refused.

    The message always carries the failing SQL, because "constraint failed"
    without the statement is not something an operator can act on.
    """


class DatabaseUnavailable(LibsqlError):
    """The database could not be reached at all."""


@dataclass
class ResultSet:
    """One statement's outcome."""

    columns: list[str] = field(default_factory=list)
    rows: list[list[Any]] = field(default_factory=list)
    rows_affected: int = 0

    def first(self) -> list[Any] | None:
        return self.rows[0] if self.rows else None

    def scalar(self, default: Any = None) -> Any:
        row = self.first()
        if row is None or not row:
            return default
        return row[0]

    def dicts(self) -> list[dict[str, Any]]:
        return [dict(zip(self.columns, row)) for row in self.rows]


class LibsqlClient(Protocol):
    """What the rest of the application is allowed to assume about a database."""

    async def execute(self, sql: str, args: Sequence[Any] = ()) -> ResultSet: ...

    async def batch(self, statements: Sequence[Statement]) -> list[ResultSet]: ...

    async def close(self) -> None: ...

# Rate limiting

class ProcessRateLimiter:
    """Serialises every database call to at most `1 / interval` per second.

    A single lock and a single timestamp, shared by everything that touches the
    database, so the limit is a property of the process rather than of each
    caller. `interval_seconds=0.5` is the two-requests-per-second ceiling.
    """

    def __init__(self, interval_seconds: float = 0.5) -> None:
        if interval_seconds < 0:
            raise ValueError("interval_seconds must not be negative")
        self.interval_seconds = float(interval_seconds)
        self._lock = asyncio.Lock()
        self._last_call = 0.0
        self._calls = 0

    @property
    def calls(self) -> int:
        """How many calls have passed through this limiter."""
        return self._calls

    async def acquire(self) -> None:
        """Wait until the next call is allowed, then claim the slot."""
        async with self._lock:
            now = time.monotonic()
            if self._last_call:
                wait_for = self.interval_seconds - (now - self._last_call)
                if wait_for > 0:
                    await asyncio.sleep(wait_for)
                    now = time.monotonic()
            self._last_call = now
            self._calls += 1

    # `wait` is the name used at the call sites; keep both spellings working.
    async def wait(self) -> None:
        await self.acquire()

_NULL = {"type": "null"}


def encode_arg(value: Any) -> dict[str, Any]:
    """Encode a Python value as a Hrana typed value.

    The wire format is asymmetric and getting it wrong is fatal: an `integer`
    carries a **decimal string** (`{"type": "integer", "value": "42"}`) while
    a `float` carries a **JSON number** (`{"type": "float", "value": 0.999}`).
    Sending a float as `"0.999"` makes Turso reject the whole pipeline with
    `JSON parse error: invalid type: string "0.999", expected f64`, which is
    what happened to every `quality_score` write before this was fixed.
    """
    if value is None:
        return dict(_NULL)
    if isinstance(value, bool):
        return {"type": "integer", "value": "1" if value else "0"}
    if isinstance(value, int):
        return {"type": "integer", "value": str(value)}
    if isinstance(value, float):
        return {"type": "float", "value": float(value)}
    if isinstance(value, Decimal):
        return {"type": "float", "value": float(value)}
    if isinstance(value, (bytes, bytearray, memoryview)):
        return {"type": "blob", "base64": base64.b64encode(bytes(value)).decode("ascii")}
    return {"type": "text", "value": str(value)}


def decode_cell(cell: Any) -> Any:
    """Decode a Hrana typed value into a Python value."""
    if not isinstance(cell, dict):
        return cell
    kind = cell.get("type")
    if kind == "null":
        return None
    if kind == "integer":
        raw = cell.get("value")
        return int(raw) if raw is not None else None
    if kind == "float":
        raw = cell.get("value")
        return float(raw) if raw is not None else None
    if kind == "blob":
        raw = cell.get("base64") or ""
        return base64.b64decode(raw)
    if kind == "text":
        return cell.get("value")
    return cell.get("value")


def rewrite_scheme(url: str) -> str:
    """Map libSQL schemes onto the HTTP client Turso actually accepts.

    `libsql://`, `ws://` and `wss://` all select a WebSocket backend in most
    clients, and Turso's Hrana endpoint rejects those with a 400. The pipeline
    endpoint is plain HTTPS.
    """
    url = url.strip()
    for prefix, replacement in (
        ("libsql://", "https://"),
        ("wss://", "https://"),
        ("ws://", "http://"),
    ):
        if url.startswith(prefix):
            return replacement + url[len(prefix) :]
    return url


class TursoHttpClient:
    """A minimal, strict client for Turso's Hrana-over-HTTP pipeline."""

    def __init__(
        self,
        url: str,
        token: str = "",
        *,
        limiter: ProcessRateLimiter | None = None,
        timeout: float = 20.0,
        transport: httpx.AsyncBaseTransport | None = None,
    ) -> None:
        if not url:
            raise ValueError("a Turso database URL is required")
        self.base_url = rewrite_scheme(url).rstrip("/")
        self.token = token
        self.limiter = limiter or ProcessRateLimiter(0.5)
        self._client = httpx.AsyncClient(
            timeout=timeout,
            transport=transport,
            headers=self._headers(),
        )

    def _headers(self) -> dict[str, str]:
        headers = {"content-type": "application/json"}
        if self.token:
            headers["authorization"] = f"Bearer {self.token}"
        return headers

    async def execute(self, sql: str, args: Sequence[Any] = ()) -> ResultSet:
        results = await self.batch([(sql, args)])
        return results[0]

    async def batch(self, statements: Sequence[Statement]) -> list[ResultSet]:
        """Run statements in order in one request and check every one of them."""
        if not statements:
            return []

        payload = {
            "requests": [
                {
                    "type": "execute",
                    "stmt": {
                        "sql": sql,
                        "args": [encode_arg(value) for value in args],
                    },
                    "want_rows": True,
                }
                for sql, args in statements
            ]
        }

        await self.limiter.acquire()
        try:
            response = await self._client.post(f"{self.base_url}/v2/pipeline", json=payload)
        except httpx.HTTPError as error:
            raise DatabaseUnavailable(f"Turso request failed: {error}") from error

        if response.status_code >= 400:
            raise DatabaseUnavailable(
                f"Turso answered HTTP {response.status_code}: {response.text[:300]}"
            )

        try:
            body = response.json()
        except ValueError as error:
            raise DatabaseUnavailable("Turso returned a non-JSON body") from error

        if "error" in body and body["error"]:
            raise LibsqlError(f"Turso rejected the pipeline: {body['error']}")

        steps = body.get("results") or []
        if len(steps) != len(statements):
            raise LibsqlError(
                f"Turso returned {len(steps)} results for {len(statements)} statements"
            )

        results: list[ResultSet] = []
        for index, (step, (sql, _args)) in enumerate(zip(steps, statements)):
            results.append(self._decode_step(step, index, sql))
        return results

    @staticmethod
    def _decode_step(step: dict[str, Any], index: int, sql: str) -> ResultSet:
        if step.get("type") == "error":
            error = step.get("error") or {}
            raise LibsqlError(
                f"libsql step {index} failed: {error.get('message', 'unknown error')} "
                f"(statement: {sql})"
            )

        response = step.get("response") or {}
        if response.get("type") == "error":
            error = response.get("error") or {}
            raise LibsqlError(
                f"libsql step {index} failed: {error.get('message', 'unknown error')} "
                f"(statement: {sql})"
            )

        result = response.get("result") or {}
        columns = [column.get("name", "") for column in result.get("cols") or []]
        rows = [[decode_cell(cell) for cell in row] for row in result.get("rows") or []]
        affected = int(result.get("affected_row_count") or 0)
        return ResultSet(columns=columns, rows=rows, rows_affected=affected)

    async def close(self) -> None:
        await self._client.aclose()

class LocalSqliteClient:
    """The same interface over the standard library, for local runs and tests."""

    def __init__(self, path: str = ":memory:") -> None:
        self.path = path
        self._connection = sqlite3.connect(path, isolation_level=None, check_same_thread=False)
        self._connection.row_factory = None
        self._lock = asyncio.Lock()
        self._connection.execute("PRAGMA journal_mode=WAL")
        self._connection.execute("PRAGMA foreign_keys=ON")

    async def execute(self, sql: str, args: Sequence[Any] = ()) -> ResultSet:
        results = await self.batch([(sql, args)])
        return results[0]

    async def batch(self, statements: Sequence[Statement]) -> list[ResultSet]:
        if not statements:
            return []
        async with self._lock:
            return await asyncio.to_thread(self._run, list(statements))

    def _run(self, statements: list[Statement]) -> list[ResultSet]:
        results: list[ResultSet] = []
        for sql, args in statements:
            try:
                cursor = self._connection.execute(sql, tuple(args))
                columns = [description[0] for description in cursor.description or []]
                rows = [list(row) for row in cursor.fetchall()] if cursor.description else []
                results.append(
                    ResultSet(
                        columns=columns,
                        rows=rows,
                        rows_affected=cursor.rowcount if cursor.rowcount > 0 else 0,
                    )
                )
            except sqlite3.Error as error:
                raise LibsqlError(f"sqlite failed: {error} (statement: {sql})") from error
        return results

    async def close(self) -> None:
        await asyncio.to_thread(self._connection.close)

def build_client(settings: Settings | None = None) -> LibsqlClient:
    """Build the client the configured URL asks for, or refuse to guess."""
    settings = settings or get_settings()
    url = settings.database_url

    if not url or url.startswith("file:"):
        if settings.is_production and not settings.allow_local_db:
            raise DatabaseUnavailable(
                "refusing to start: no remote database configured. Set "
                "TURSO_DB_URL and TURSO_AUTH_TOKEN, or set ALLOW_LOCAL_DB=true "
                "if a local file is genuinely intended."
            )
        path = url[len("file:") :] if url.startswith("file:") else settings.local_db_path
        path = path or ":memory:"
        logger.warning(
            "Connected to LOCAL SQLite: %s — this is not the Turso cloud database", path
        )
        return LocalSqliteClient(path)

    if not settings.turso_auth_token:
        raise DatabaseUnavailable(
            f"TURSO_AUTH_TOKEN is required for the remote database {url}"
        )

    logger.info("Connected to Remote Turso DB: %s (HttpClient)", rewrite_scheme(url))
    return TursoHttpClient(
        url,
        settings.turso_auth_token,
        limiter=ProcessRateLimiter(settings.libsql_min_interval_seconds),
    )

"""Turso (libSQL) schema bootstrap.

Every table is `CREATE ... IF NOT EXISTS`, so a cold start against an empty
database builds the whole thing with no manual step, and a warm start is a
no-op.

`page_fts` is **external content** over `pages`: the text is not stored twice,
and three triggers keep the index in step with the table. Those triggers are
not optional decoration — without them an external-content FTS5 table has no
rows to search, and every query returns nothing while every write reports
success.
"""

    CREATE VIRTUAL TABLE IF NOT EXISTS page_fts USING fts5(

    )
    """,

    CREATE TABLE IF NOT EXISTS crawl_queue (

        status TEXT DEFAULT 'pending',

    CREATE TRIGGER IF NOT EXISTS pages_ai AFTER INSERT ON pages BEGIN

        VALUES (new.page_id, new.title, new.description, new.body_text);
    END
    """,
    """
    CREATE TRIGGER IF NOT EXISTS pages_ad AFTER DELETE ON pages BEGIN
        INSERT INTO page_fts(page_fts, rowid, title, description, body_text)

    END
    """,
    """
    CREATE TRIGGER IF NOT EXISTS pages_au AFTER UPDATE ON pages BEGIN
        INSERT INTO page_fts(page_fts, rowid, title, description, body_text)
        VALUES ('delete', old.page_id, old.title, old.description, old.body_text);
        INSERT INTO page_fts(rowid, title, description, body_text)

    END
    """,

def _scalar(result: ResultSet, default: int = 0) -> int:
    value = result.scalar(default)
    try:
        return int(value)
    except (TypeError, ValueError):
        return default


#: Tables only the pre-rebuild backend created. Any of these present means the
#: database predates this schema.
LEGACY_TABLES: tuple[str, ...] = (
    "documents_fts",
    "documents",
    "search_records",
    "crawl_backlog",
    "page_fts",
    "pages",
    "sites",
    "crawl_queue",
)

class LegacySchemaError(RuntimeError):
    """The database holds a pre-rebuild schema and dropping it was not allowed."""


async def _table_columns(client: LibsqlClient, table: str) -> set[str]:
    """The column names of a table, or an empty set when it does not exist."""
    try:
        result = await client.execute(f"PRAGMA table_info({table})")
    except LibsqlError:
        return set()
    return {str(row[1]) for row in result.rows if len(row) > 1}


async def detect_legacy_schema(client: LibsqlClient) -> list[str]:
    """Name the legacy artefacts this database still holds, if any.

    `sites` and `crawl_queue` exist in both schemas, so their presence proves
    nothing on its own. The signals that do are a `pages` table without
    `body_text`, and the tables only the previous backend created.
    """
    found: list[str] = []

    page_columns = await _table_columns(client, "pages")
    legacy_pages = bool(page_columns) and "body_text" not in page_columns
    if legacy_pages:
        found.append("pages (pre-rebuild columns)")

    for table in ("documents", "documents_fts", "search_records", "crawl_backlog"):
        if await _table_columns(client, table):
            found.append(table)

    # These three belong to the old schema too, but only count as evidence
    # once the old `pages` shape or an old-only table has confirmed it.
    if legacy_pages:
        for table in ("page_fts", "sites", "crawl_queue"):
            if await _table_columns(client, table):
                found.append(table)

    return found


async def migrate_legacy_schema(
    client: LibsqlClient,
    *,
    allow_drop: bool = True,
) -> list[str]:
    """Clear a pre-rebuild schema so the new one can be created.

    `CREATE TABLE IF NOT EXISTS` cannot change an existing table, so a database
    left over from the previous backend would keep its old columns and every
    write would fail with a missing-column error. The old rows are a crawl
    cache in a shape this code cannot read, so they are dropped rather than
    migrated, and what was removed is logged.
    """
    found = await detect_legacy_schema(client)
    if not found:
        return []

    if not allow_drop:
        raise LegacySchemaError(
            "the database still holds the pre-rebuild schema ("
            + ", ".join(found)
            + "). Set MIGRATE_LEGACY_SCHEMA=true to drop it, or point the app at "
            "an empty database."
        )

    logger.warning(
        "pre-rebuild schema detected (%s) — dropping it so the new schema can be created",
        ", ".join(found),
    )

    statements: list[tuple[str, tuple]] = [("BEGIN IMMEDIATE", ())]
    for table in LEGACY_TABLES:
        statements.append((f"DROP TABLE IF EXISTS {table}", ()))
    statements.append(("COMMIT", ()))

    try:
        await client.batch(statements)
    except LibsqlError as error:
        logger.error("legacy schema drop failed: %s", error)
        try:
            await client.execute("ROLLBACK")
        except LibsqlError:
            pass
        raise

    logger.warning("legacy schema dropped: %s", ", ".join(found))
    return found


async def initialize_database(
    client: LibsqlClient,
    *,
    migrate: bool = True,
    allow_legacy_drop: bool = True,
) -> dict[str, int]:
    """Create every table, trigger and index; return what the index holds."""
    if migrate:
        await migrate_legacy_schema(client, allow_drop=allow_legacy_drop)

    await client.batch([(sql, ()) for sql in SCHEMA_STATEMENTS])
    await ensure_columns(client)
    await client.batch([(sql, ()) for sql in POST_COLUMN_STATEMENTS])
    counts = await read_counts(client)
    logger.info(
        "schema ready: %d sites, %d pages, %d queued",
        counts["sites"],
        counts["pages"],
        counts["queued"],
    )
    return counts


async def ensure_columns(client: LibsqlClient) -> list[str]:
    """Add any column a later build introduced to an existing table."""
    added: list[str] = []

    for table, column, declaration in COLUMN_ADDITIONS:
        columns = await _table_columns(client, table)
        if not columns or column in columns:
            continue
        await client.execute(f"ALTER TABLE {table} ADD COLUMN {column} {declaration}")
        added.append(f"{table}.{column}")

    if added:
        logger.info("migration applied: %s", ", ".join(added))
    return added


async def read_counts(client: LibsqlClient) -> dict[str, int]:
    """The three integers the health and stats endpoints report."""
    results = await client.batch([(sql, ()) for sql in COUNT_STATEMENTS])
    return {
        "pages": _scalar(results[0]),
        "sites": _scalar(results[1]),
        "queued": _scalar(results[2]),
    }


async def verify_writable(client: LibsqlClient) -> bool:
    """Prove the database accepts writes before a crawl relies on it.

    A read-only token, the wrong branch or a staging database fails here,
    loudly, instead of silently during a crawl. The sentinel row is removed
    again, so the check leaves the index exactly as it found it.
    """
    sentinel = "https://verify.eyesearch.invalid/__writable_probe__"
    try:
        await client.batch(
            [
                ("BEGIN IMMEDIATE", ()),
                ("DELETE FROM pages WHERE url = ?", (sentinel,)),
                (
                    "INSERT INTO pages (site_id, url, title, description, body_text, digest) "
                    "VALUES (NULL, ?, ?, ?, ?, ?)",
                    (sentinel, "probe", "probe", "probe", "probe"),
                ),
                ("DELETE FROM pages WHERE url = ?", (sentinel,)),
                ("COMMIT", ()),
            ]
        )
    except LibsqlError as error:
        logger.error("database is not writable: %s", error)
        try:
            await client.execute("ROLLBACK")
        except LibsqlError:
            pass
        return False
    return True


async def probe(client: LibsqlClient) -> dict[str, Any]:
    """A one-shot health read used by `/health` and `--probe`."""
    try:
        counts = await read_counts(client)
    except LibsqlError as error:
        return {"db": "down", "error": str(error)}
    return {"db": "up", **counts}

"""The pipeline: one shared HTTP client, six agents, one batch loop.

`Orchestrator` is the only place the agents meet. It owns the pooled
`httpx.AsyncClient`, pulls a batch of ten URLs from the frontier, runs them
through fetch → parse → quality → index, and closes or fails each queue row as
its own outcome dictates.

The counters it reports are honest by construction: `pages` comes from the
store's confirmed commit, never from the number of documents it handed over.
"""

@dataclass
class BatchOutcome:
    """What one batch of URLs achieved."""

    attempted: int = 0
    fetched: int = 0
    parsed: int = 0
    accepted: int = 0
    rejected: int = 0
    indexed_pages: int = 0
    queue_closed: int = 0
    queue_failed: int = 0
    skipped: int = 0
    discovered: int = 0
    paused: bool = False
    skip_reasons: dict[str, int] = field(default_factory=dict)
    elapsed_ms: int = 0

    @property
    def empty(self) -> bool:
        return self.attempted == 0

    def as_dict(self) -> dict[str, object]:
        return {
            "attempted": self.attempted,
            "fetched": self.fetched,
            "parsed": self.parsed,
            "accepted": self.accepted,
            "rejected": self.rejected,
            "indexed_pages": self.indexed_pages,
            "queue_closed": self.queue_closed,
            "queue_failed": self.queue_failed,
            "skipped": self.skipped,
            "discovered": self.discovered,
            "paused": self.paused,
            "skip_reasons": dict(self.skip_reasons),
            "elapsed_ms": self.elapsed_ms,
        }


class Orchestrator:
    """Wires the agents together over one client and one store."""

    def __init__(
        self,
        client: LibsqlClient,
        settings: Settings | None = None,
        transport: httpx.AsyncBaseTransport | None = None,
    ) -> None:
        self._settings = settings or get_settings()
        self._store = IndexStore(client, self._settings)
        self._guard = StorageGuard(
            client,
            limit=self._settings.storage_page_limit,
            ttl_seconds=self._settings.storage_count_ttl_seconds,
        )
        self._fetcher = FetcherAgent(self._settings, transport=transport)
        self._parser = ParserAgent()
        self._quality = QualityAgent()
        self._classifier = ClassifierAgent()
        self._language = LanguageAgent()
        self._indexer = IndexerAgent(self._store, self._guard)
        self._frontier = FrontierAgent(self._store, self._settings)
        self._revalidation = RevalidationAgent(
            self._store,
            self._fetcher,
            self._settings,
            parser=self._parser,
        )
        self.batches = 0
        self.pages_indexed = 0

    # -- accessors ---------------------------------------------------------

    @property
    def store(self) -> IndexStore:
        return self._store

    @property
    def guard(self) -> StorageGuard:
        return self._guard

    @property
    def frontier(self) -> object:
        return self._frontier

    @property
    def fetcher(self) -> object:
        return self._fetcher

    @property
    def revalidation(self) -> object:
        return self._revalidation

    # -- one batch ---------------------------------------------------------

    async def run_batch(self, limit: int | None = None) -> BatchOutcome:
        """Fetch, parse, gate and store one batch of queued URLs."""
        started = time.monotonic()
        outcome = BatchOutcome()

        queue: list[QueuedUrl] = await self._frontier.next_batch(limit)
        if not queue:
            return outcome

        outcome.attempted = len(queue)

        # One storage read for the whole batch — never one per URL.
        state = await self._guard.storage_state()
        if state.paused:
            outcome.paused = True
            logger.warning("batch skipped, storage paused: %s", state.reason)
            return outcome

        documents: list[PageDocument] = []
        queue_ids: list[int] = []
        failed_ids: list[int] = []
        skipped_ids: list[int] = []
        discovered: list[str] = []

        # Per-domain page cap: one count query for the whole batch, and URLs on
        # a domain that is already full are parked before any fetch happens.
        cap = self._settings.max_pages_per_domain
        counts = await self._store.domain_page_counts([item.domain for item in queue])
        eligible: list[QueuedUrl] = []
        for item in queue:
            if counts.get(item.domain, 0) >= cap:
                outcome.skipped += 1
                self._tally(outcome.skip_reasons, "domain_cap")
                skipped_ids.append(item.rowid)
            else:
                eligible.append(item)

        self._quality.reset()
        results = await self._fetcher.fetch_many([item.url for item in eligible])

        for item, result in zip(eligible, results):
            # A domain can fill up part-way through a batch.
            if counts.get(item.domain, 0) >= cap:
                outcome.skipped += 1
                self._tally(outcome.skip_reasons, "domain_cap")
                skipped_ids.append(item.rowid)
                continue

            if not result.ok:
                outcome.rejected += 1
                self._tally(outcome.skip_reasons, result.reason or "fetch_failed")
                failed_ids.append(item.rowid)
                continue

            outcome.fetched += 1
            parsed = self._parser.parse(item.url, result.html)
            if not parsed.ok:
                outcome.rejected += 1
                self._tally(outcome.skip_reasons, parsed.reason or "parse_failed")
                failed_ids.append(item.rowid)
                continue

            outcome.parsed += 1

            # Every page that parsed contributes its outbound links, whether
            # or not it passed the gate: a thin page can still point at good
            # ones. The frontier applies the crawl-scope rules.
            discovered.extend(self._frontier.filter_links(parsed.links, item.url))

            document = PageDocument(
                url=item.url,
                domain=item.domain,
                title=parsed.title,
                description=parsed.description,
                body_text=parsed.body_text,
            )
            verdict = self._quality.check(document)
            if not verdict.ok:
                outcome.rejected += 1
                self._tally(outcome.skip_reasons, verdict.reason or "rejected")
                failed_ids.append(item.rowid)
                continue

            document.digest = verdict.digest
            document.quality_score = verdict.score

            # Tagged here, at ingestion, so a search can filter without a
            # backfill later.
            document.category = self._classifier.classify(
                item.url,
                title=document.title,
                description=document.description,
                body_text=document.body_text,
            ).category
            document.language = self._language.detect(
                f"{document.title} {document.body_text}"
            )

            documents.append(document)
            queue_ids.append(item.rowid)
            counts[item.domain] = counts.get(item.domain, 0) + 1

        outcome.accepted = len(documents)

        if failed_ids:
            outcome.queue_failed = await self._indexer.mark_failed(
                failed_ids, reason="fetch, parse or quality gate rejected the page"
            )

        if skipped_ids:
            await self._indexer.mark_skipped(
                skipped_ids, reason=f"domain reached its {cap}-page cap"
            )

        if documents:
            index_outcome = await self._indexer.persist(documents, queue_ids)
            if index_outcome.committed:
                # Only a confirmed commit may move the counters.
                outcome.indexed_pages = index_outcome.pages_written
                outcome.queue_closed = index_outcome.queue_closed
                self.pages_indexed += index_outcome.counted
                self.batches += 1
            else:
                outcome.paused = index_outcome.paused
                logger.warning("batch not persisted: %s", index_outcome.reason)

        if discovered and not outcome.paused:
            # Do not grow the frontier with URLs whose domain is already full.
            discovered = [
                url
                for url in discovered
                if counts.get(_frontier_module.domain_of(url), 0) < cap
            ]

        if discovered and not outcome.paused:
            # Queue after the commit, so a failed write never grows the frontier.
            outcome.discovered = await self._frontier.enqueue(discovered)

        outcome.elapsed_ms = int((time.monotonic() - started) * 1000)
        logger.info(
            "batch done: %d queued, %d fetched, %d accepted, %d rejected, "
            "%d skipped, %d indexed, %d discovered (%dms)",
            outcome.attempted,
            outcome.fetched,
            outcome.accepted,
            outcome.rejected,
            outcome.skipped,
            outcome.indexed_pages,
            outcome.discovered,
            outcome.elapsed_ms,
        )
        return outcome

    # -- the loop ----------------------------------------------------------

    async def run_forever(self) -> None:
        """Consume the frontier until cancelled.

        2.0 s between batches; a 15.0 s non-blocking backoff when the queue is
        empty, so an idle deployment costs nothing and still recovers the
        moment there is work.
        """
        while True:
            try:
                outcome = await self.run_batch()
            except asyncio.CancelledError:
                raise
            except Exception as error:  # a failed batch must not end the loop
                logger.exception("batch failed: %s", error)
                outcome = None

            if outcome is not None and not outcome.empty and not outcome.paused:
                await asyncio.sleep(self._settings.crawl_batch_sleep_seconds)
                continue

            delay = self._settings.crawl_empty_backoff_seconds
            if outcome is not None and outcome.empty:
                added = await self._frontier.refill(minimum=1)
                if not added:
                    # Nothing to crawl and nothing past its cooldown yet.
                    delay = max(delay, IDLE_REFILL_FAILED_BACKOFF_SECONDS)

            logger.debug("crawl worker sleeping %.1fs (queue empty)", delay)
            await asyncio.sleep(delay)

    async def run_domains(self, domains: Sequence[str]) -> BatchOutcome:
        """Seed specific domains and process exactly one batch of them."""
        await self._frontier.enqueue([f"https://{domain}/" for domain in domains])
        return await self.run_batch(limit=len(domains) or self._settings.crawl_batch_size)

    async def close(self) -> None:
        await self._fetcher.aclose()

    @staticmethod
    def _tally(counter: dict[str, int], reason: str) -> None:
        counter[reason] = counter.get(reason, 0) + 1

@asynccontextmanager
async def lifespan(app: FastAPI):
    settings = get_settings()
    logging.basicConfig(
        level=getattr(logging, settings.log_level.upper(), logging.INFO),
        format="%(asctime)s %(levelname)s %(name)s %(message)s",

    logger.info("database target: %s", describe_database_target(settings))

    client: LibsqlClient = build_client(settings)
    await initialize_database(client, allow_legacy_drop=settings.migrate_legacy_schema)
    writable = await verify_writable(client)
    if not writable:
        logger.error("database rejected the write probe — crawling will stay paused")

    orchestrator = Orchestrator(client, settings)
    app.state.settings = settings
    app.state.client = client
    app.state.orchestrator = orchestrator
    app.state.service = SearchService(orchestrator.store, settings)
    app.state.workers = WorkerHandles()

    # Workers are scheduled from a bootstrap task, not inline, so the port is
    # bound and /health answers before the first crawl begins.
    bootstrap = asyncio.create_task(_start_workers_when_serving(app, orchestrator, settings))
    app.state.bootstrap = bootstrap

    try:
        yield
    finally:
        bootstrap.cancel()
        await stop_workers(app.state.workers)
        await orchestrator.close()
        await client.close()
        logger.info("shutdown complete")

def compose_answer(query: str, results: list[dict[str, Any]]) -> str:
    """An extractive answer assembled only from indexed text.

    There is no model behind this endpoint, so it does not invent prose: it
    reports what the index holds and attributes each sentence to its source.
    """
    if not results:
        return (
            f"Nothing is indexed for **{query}** yet. The query has been left to the "
            "crawler; ask again once it has been fetched."
        )

    lines = [f"Here is what the index holds for **{query}**.", ""]
    for result in results[:3]:
        title = result.get("title") or result.get("url", "")
        snippet = (result.get("snippet") or "").strip()
        lines.append(f"**{title}**")
        if snippet:
            lines.append(snippet)
        lines.append(f"Source: {result.get('url', '')}")
        lines.append("")
    return "\n".join(lines).strip()


def sse(event: dict[str, Any]) -> str:
    return f"data: {json.dumps(event, ensure_ascii=False)}\n\n"


@app.post("/v1/chat")
async def chat(request: Request, body: ChatRequest) -> StreamingResponse:
    query = require_query(body.query)
    service = get_service(request)
    payload = await service.search(query, limit=5)
    results = payload.get("results", [])
    answer = compose_answer(query, results)
    sources = [{"title": item["title"], "url": item["url"]} for item in results[:5]]

    async def stream() -> AsyncIterator[str]:
        buffer: list[str] = []
        for token in answer.split(" "):
            buffer.append(token)
            if len(buffer) >= 6:
                yield sse({"delta": " ".join(buffer) + " "})
                buffer = []
                await asyncio.sleep(0)
        if buffer:
            yield sse({"delta": " ".join(buffer)})
        yield sse({"done": True, "sources": sources})

    return StreamingResponse(
        stream(),
        media_type="text/event-stream",
        headers={"cache-control": "no-cache", "x-accel-buffering": "no"},
    )

class _ExplodingClient:
    """Fails the batch at the page insert, after the transaction opened."""

    def __init__(self, inner, marker: str = "INSERT OR IGNORE INTO pages") -> None:
        self.inner = inner
        self.marker = marker

    async def execute(self, sql, args=()):
        return await self.inner.execute(sql, args)

    async def batch(self, statements):
        for sql, _args in statements:
            if self.marker in sql:
                raise LibsqlError(f"boom (statement: {sql})")
        return await self.inner.batch(statements)

    async def close(self):
        await self.inner.close()

"""Match snippets and title ranking.

The brief asked for FTS5's `snippet()`. On the SQLite build this project runs
on, `snippet()` ignores its `colnum` argument and returns column 0 for every
call — `highlight()` behaves correctly, `snippet()` does not. The window is
therefore computed in Python, and these tests pin the behaviour that matters:
the passage shown contains the term that matched, and a title match is not
echoed back as the snippet.
"""

from __future__ import annotations

import app.engine.store as store_module
from app.engine.store import PageDocument, passage_around_match

FILLER = "The department publishes routine notices for district offices. " * 6


def _doc(url: str, title: str, body: str, digest: str) -> PageDocument:
    return PageDocument(
        url=url,
        domain=url.split("//", 1)[1].split("/", 1)[0],
        title=title,
        description=f"{title} description.",
        body_text=body,
        digest=digest,
    )


# ---------------------------------------------------------------------------
# The passage window
# ---------------------------------------------------------------------------


def test_the_passage_contains_the_term_that_matched():
    body = FILLER + "Bengaluru metro fares were revised in March." + FILLER

    passage = passage_around_match(body, ["metro"])

    assert "metro" in passage.lower()


def test_a_passage_is_shorter_than_a_long_body():
    body = FILLER * 5 + "metro" + FILLER * 5

    assert len(passage_around_match(body, ["metro"])) < len(body)


def test_no_match_returns_nothing_so_the_caller_can_fall_back():
    assert passage_around_match("nothing relevant here at all", ["metro"]) == ""
    assert passage_around_match("", ["metro"]) == ""
    assert passage_around_match("some text", []) == ""


def test_the_window_does_not_cut_a_word_in_half():
    passage = passage_around_match("alpha beta gamma delta", ["gamma"], window=12)

    assert passage == "…gamma…"


def test_a_match_at_the_start_has_no_leading_ellipsis():
    passage = passage_around_match("metro fares rose today and more follows", ["metro"], window=20)

    assert not passage.startswith("…")
    assert passage.startswith("metro")


def test_the_first_occurrence_wins():
    body = "alpha metro beta " + ("filler " * 100) + "metro again"

    passage = passage_around_match(body, ["metro"], window=40)

    assert passage.count("metro") == 1

async def _seed_title_and_body(store) -> None:
    await store.persist_batch(
        [
            _doc("https://title.gov.in/", "Metro fares revised", "Fares went up this year.", "title"),
            _doc("https://body.gov.in/", "Notices", "Metro fares were revised this year.", "body"),
        ]
    )


async def test_a_title_match_ranks_first(store):
    await _seed_title_and_body(store)

    hits = await store.search("metro")

    assert hits[0].title == "Metro fares revised"


async def test_the_title_boost_lowers_the_ranking_key(store, monkeypatch):
    """bm25 is lower-is-better, so a boost must make the key more negative."""
    await _seed_title_and_body(store)

    boosted = await store.search("metro")
    boosted_key = next(hit.raw_score for hit in boosted if "Metro" in hit.title)

    monkeypatch.setattr(store_module, "TITLE_MATCH_BOOST", 0.0)
    plain = await store.search("metro")
    plain_key = next(hit.raw_score for hit in plain if "Metro" in hit.title)

    assert boosted_key < plain_key


async def test_the_reported_score_is_monotone_with_the_order(store):
    await _seed_title_and_body(store)

    hits = await store.search("metro")
    scores = [hit.score for hit in hits]

    assert scores == sorted(scores, reverse=True)
    assert scores[0] == 1.0


async def test_the_boost_does_not_change_which_pages_match(store):
    await _seed_title_and_body(store)

    hits = await store.search("metro")

    assert len(hits) == 2


async def test_filters_still_apply_with_the_boost(store):
    first = _doc("https://a.gov.in/", "Metro fares", "Fares.", "a")
    first.category = "government"
    second = _doc("https://b.gov.in/", "Metro news", "News.", "b")
    second.category = "news_media"
    await store.persist_batch([first, second])

    hits = await store.search("metro", category="news_media")

    assert [hit.title for hit in hits] == ["Metro news"]
